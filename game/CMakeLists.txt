# Web app (Emscripten): SDL2 comes from the Emscripten port; assets are preloaded from RN_WEB_ASSETS (a staging folder of the user's extracted data) for local testing
option(RN_WEB_APP "Build the browser app with Emscripten" OFF)
set(RN_WEB_ASSETS "" CACHE PATH "folder preloaded into the web app's virtual file system as /assets")
if(EMSCRIPTEN AND RN_WEB_APP)
  include(FetchContent)
  FetchContent_Declare(stbvorbis URL "${RN_VORBIS_SOURCE}"
    URL_HASH SHA256=4c7cb2ff1f7011e9d67950446b7eb9ca044f2e464d76bfbb0b84dd2e23e65636 DOWNLOAD_NO_EXTRACT TRUE)
  FetchContent_MakeAvailable(stbvorbis)
  add_executable(ccr app/main.cpp)
  target_include_directories(ccr PRIVATE ${stbvorbis_SOURCE_DIR})
  target_compile_options(ccr PRIVATE -sUSE_SDL=2)
  target_link_options(ccr PRIVATE -sUSE_SDL=2 -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=268435456 -sEXIT_RUNTIME=0 "-sEXPORTED_RUNTIME_METHODS=['FS','callMain']")
  if(RN_WEB_ASSETS)
    target_link_options(ccr PRIVATE "--preload-file" "${RN_WEB_ASSETS}@/assets")
  endif()
  target_link_options(ccr PRIVATE "--shell-file" "${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html")
  set_target_properties(ccr PROPERTIES SUFFIX ".html")
  add_custom_command(TARGET ccr POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_SOURCE_DIR}/web/manifest.webmanifest ${CMAKE_CURRENT_SOURCE_DIR}/web/sw.js $<TARGET_FILE_DIR:ccr>)
endif()

# Development viewer (SDL2, fetched and built statically)
option(RN_BUILD_VIEWER "Build the SDL2 development viewer" ON)
if(RN_BUILD_VIEWER AND NOT EMSCRIPTEN)
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL2 URL "${RN_SDL_ARCHIVE}"
    URL_HASH SHA256=24b574f71c87a763f50704bbb630cbe38298d544a1f890f099a4696b1d6beba4)
  FetchContent_MakeAvailable(SDL2)
  FetchContent_Declare(stbvorbis URL "${RN_VORBIS_SOURCE}"
    URL_HASH SHA256=4c7cb2ff1f7011e9d67950446b7eb9ca044f2e464d76bfbb0b84dd2e23e65636 DOWNLOAD_NO_EXTRACT TRUE)
  FetchContent_MakeAvailable(stbvorbis)
  add_executable(ccr app/main.cpp)
  target_include_directories(ccr PRIVATE ${stbvorbis_SOURCE_DIR})
  target_link_libraries(ccr PRIVATE SDL2::SDL2-static)
  if(MSVC)
    target_compile_options(ccr PRIVATE /W3 /fp:precise)
  endif()
  add_executable(viewer viewer/main.cpp)
  target_link_libraries(viewer PRIVATE SDL2::SDL2-static)
  if(MSVC)
    target_compile_options(viewer PRIVATE /W3 /fp:precise)
  endif()
endif()
