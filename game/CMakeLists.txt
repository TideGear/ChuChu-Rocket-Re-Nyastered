# Development viewer (SDL2, fetched and built statically)
option(RN_BUILD_VIEWER "Build the SDL2 development viewer" ON)
if(RN_BUILD_VIEWER)
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL2 URL https://github.com/libsdl-org/SDL/releases/download/release-2.30.9/SDL2-2.30.9.tar.gz)
  FetchContent_MakeAvailable(SDL2)
  add_executable(ccr app/main.cpp)
  target_link_libraries(ccr PRIVATE SDL2::SDL2-static)
  if(MSVC)
    target_compile_options(ccr PRIVATE /W3 /fp:precise)
  endif()
  add_executable(viewer viewer/main.cpp)
  target_link_libraries(viewer PRIVATE SDL2::SDL2-static)
  if(MSVC)
    target_compile_options(viewer PRIVATE /W3 /fp:precise)
  endif()
endif()
